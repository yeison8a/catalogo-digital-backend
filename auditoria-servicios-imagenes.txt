package catalogo_digital_backend.service;

import catalogo_digital_backend.entity.ImageDeletionJob;
import catalogo_digital_backend.repository.ImageDeletionJobRepository;
import catalogo_digital_backend.storage.CloudinaryImageStorageService;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.context.annotation.Profile;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;
import org.springframework.transaction.support.TransactionTemplate;

import java.time.Instant;
import java.util.List;
import java.util.UUID;

@Service
@Profile("local")
@ConditionalOnProperty(
        name = "app.image-deletion.enabled",
        havingValue = "true"
)
public class ImageDeletionWorker {

    private static final Logger logger =
            LoggerFactory.getLogger(ImageDeletionWorker.class);

    private final ImageDeletionJobRepository jobRepository;
    private final ProductImageService imageService;
    private final CloudinaryImageStorageService cloudinaryStorage;
    private final TransactionTemplate transactionTemplate;

    public ImageDeletionWorker(
            ImageDeletionJobRepository jobRepository,
            ProductImageService imageService,
            CloudinaryImageStorageService cloudinaryStorage,
            TransactionTemplate transactionTemplate
    ) {
        this.jobRepository = jobRepository;
        this.imageService = imageService;
        this.cloudinaryStorage = cloudinaryStorage;
        this.transactionTemplate = transactionTemplate;
    }

    // Buscar trabajos pendientes periódicamente.
    @Scheduled(
            fixedDelayString =
                    "${app.image-deletion.poll-ms:15000}"
    )
    public void processPendingJobs() {

        List<ImageDeletionJob> jobs = jobRepository
                .findTop20ByNextAttemptAtLessThanEqualOrderByNextAttemptAtAsc(
                        Instant.now()
                );

        for (ImageDeletionJob job : jobs) {
            processJob(job);
        }
    }

    // Procesar un trabajo individual.
    public void processJob(ImageDeletionJob job) {

        UUID jobId = job.getId();
        String publicId = job.getCloudinaryPublicId();

        try {

            // Paso 1: eliminar el recurso remoto.
            // Las imágenes antiguas no tienen publicId.
            if (publicId != null && !publicId.isBlank()) {
                cloudinaryStorage.delete(publicId);
            }

            // Paso 2: confirmar la eliminación local.
            // La imagen y el trabajo se eliminan
            // dentro de la misma transacción.
            transactionTemplate.executeWithoutResult(status -> {

                if (!jobRepository.existsById(jobId)) {
                    return;
                }

                imageService.deleteImage(
                        job.getProductId(),
                        job.getImageId()
                );

                jobRepository.deleteById(jobId);
            });

            logger.info(
                    "Trabajo de eliminación completado: {}",
                    jobId
            );

        } catch (RuntimeException exception) {

            logger.warn(
                    "Error procesando eliminación {}. Se reintentará.",
                    jobId,
                    exception
            );

            try {
                scheduleRetry(jobId);
            } catch (RuntimeException retryException) {
                logger.error(
                        "No se pudo actualizar el reintento {}",
                        jobId,
                        retryException
                );
            }
        }
    }

    private void scheduleRetry(UUID jobId) {

        transactionTemplate.executeWithoutResult(status -> {

            ImageDeletionJob current = jobRepository
                    .findById(jobId)
                    .orElse(null);

            if (current == null) {
                return;
            }

            // Retroceso exponencial, limitado a 5 minutos.
            int exponent = Math.min(
                    current.getAttempts(),
                    6
            );

            long delaySeconds = Math.min(
                    300L,
                    5L * (1L << exponent)
            );

            current.scheduleRetry(
                    Instant.now().plusSeconds(delaySeconds),
                    "El servicio externo o la base de datos "
                            + "no completó la eliminación."
            );

            jobRepository.save(current);
        });
    }
}

package catalogo_digital_backend.service;

import catalogo_digital_backend.entity.ImageDeletionJob;
import catalogo_digital_backend.entity.ProductImage;

import catalogo_digital_backend.repository.ImageDeletionJobRepository;
import catalogo_digital_backend.repository.ProductImageRepository;
import catalogo_digital_backend.repository.ProductRepository;

import org.springframework.context.annotation.Profile;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.NoSuchElementException;
import java.util.UUID;

@Service
@Profile("local")
public class ProductImageDeletionService {

    private final ProductRepository productRepository;
    private final ProductImageRepository imageRepository;
    private final ImageDeletionJobRepository jobRepository;

    public ProductImageDeletionService(
            ProductRepository productRepository,
            ProductImageRepository imageRepository,
            ImageDeletionJobRepository jobRepository
    ) {
        this.productRepository = productRepository;
        this.imageRepository = imageRepository;
        this.jobRepository = jobRepository;
    }

    @Transactional
    public void requestDeletion(
            UUID productId,
            UUID imageId
    ) {

        if (productId == null || imageId == null) {
            throw new IllegalArgumentException(
                    "Los identificadores son obligatorios."
            );
        }

        // Bloquear el producto para coordinar
        // cambios de la galería.
        productRepository.findByIdForUpdate(productId)
                .orElseThrow(
                        () -> new NoSuchElementException(
                                "No existe el producto."
                        )
                );

        ProductImage image = imageRepository
                .findById(imageId)
                .orElseThrow(
                        () -> new NoSuchElementException(
                                "No existe la imagen."
                        )
                );

        if (!image.getProduct().getId().equals(productId)) {
            throw new NoSuchElementException(
                    "La imagen no pertenece al producto."
            );
        }

        // Solicitud repetida: no crear otro trabajo.
        if (image.isDeletionPending()) {
            return;
        }

        if (jobRepository.existsByImageId(imageId)) {
            throw new IllegalStateException(
                    "Ya existe un trabajo de eliminación."
            );
        }

        // Marcar la fotografía.
        image.setDeletionPending(true);
        imageRepository.save(image);

        // Crear el trabajo en la misma transacción.
        ImageDeletionJob job = new ImageDeletionJob(
                productId,
                imageId,
                image.getCloudinaryPublicId()
        );

        jobRepository.save(job);
    }
}

package catalogo_digital_backend.service;

import catalogo_digital_backend.dto.product.image.ProductImageResponse;
import catalogo_digital_backend.storage.CloudinaryImageStorageService;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import org.springframework.context.annotation.Profile;
import org.springframework.stereotype.Service;
import org.springframework.transaction.support.TransactionTemplate;
import org.springframework.web.multipart.MultipartFile;

import java.util.UUID;

@Service
@Profile("local")
public class ProductImageUploadService {

    private static final Logger logger =
            LoggerFactory.getLogger(ProductImageUploadService.class);

    private static final int MAX_IMAGES = 5;

    private final CloudinaryImageStorageService cloudinaryStorage;
    private final ProductImageService imageService;
    private final ProductService productService;
    private final TransactionTemplate transactionTemplate;

    public ProductImageUploadService(
            CloudinaryImageStorageService cloudinaryStorage,
            ProductImageService imageService,
            ProductService productService,
            TransactionTemplate transactionTemplate
    ) {
        this.cloudinaryStorage = cloudinaryStorage;
        this.imageService = imageService;
        this.productService = productService;
        this.transactionTemplate = transactionTemplate;
    }

    public ProductImageResponse upload(
            UUID productId,
            MultipartFile file
    ) {

        // Verificar que el producto exista.
        productService.findById(productId);

        // Comprobar el límite de fotografías.
        if (imageService.findByProduct(productId).size()
                >= MAX_IMAGES) {
            throw new IllegalStateException(
                    "Un producto no puede tener más de 5 imágenes."
            );
        }

        // Subir la fotografía a Cloudinary.
        CloudinaryImageStorageService.StoredImage stored =
                cloudinaryStorage.store(file);

        try {

            // Registrar la fotografía en PostgreSQL.
            ProductImageResponse response =
                    transactionTemplate.execute(status ->
                            imageService.addUploadedImage(
                                    productId,
                                    stored.secureUrl(),
                                    stored.publicId()
                            )
                    );

            if (response == null) {
                throw new IllegalStateException(
                        "No se pudo registrar la imagen."
                );
            }

            return response;

        } catch (RuntimeException databaseException) {

            // Intentar eliminar el archivo si falla PostgreSQL.
            try {
                cloudinaryStorage.delete(stored.publicId());

            } catch (RuntimeException cleanupException) {

                logger.error(
                        "No se pudo eliminar la imagen de Cloudinary: {}",
                        stored.publicId(),
                        cleanupException
                );

                databaseException.addSuppressed(
                        cleanupException
                );
            }

            throw databaseException;
        }
    }
}

package catalogo_digital_backend.storage;

import com.cloudinary.Cloudinary;
import com.cloudinary.utils.ObjectUtils;

import org.springframework.context.annotation.Profile;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import javax.imageio.ImageIO;
import javax.imageio.ImageReader;
import javax.imageio.stream.ImageInputStream;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.util.Iterator;
import java.util.Map;
import java.util.UUID;

@Service
@Profile("local")
public class CloudinaryImageStorageService {

    private static final long MAX_FILE_SIZE =
            5L * 1024 * 1024;

    private static final int MAX_DIMENSION = 5000;

    private final Cloudinary cloudinary;

    public CloudinaryImageStorageService(
            Cloudinary cloudinary
    ) {
        this.cloudinary = cloudinary;
    }

    // 1. Subir una imagen a Cloudinary
    public StoredImage store(MultipartFile file) {

        if (file == null || file.isEmpty()) {
            throw new IllegalArgumentException(
                    "Debes seleccionar una imagen."
            );
        }

        if (file.getSize() > MAX_FILE_SIZE) {
            throw new IllegalArgumentException(
                    "La imagen no puede superar 5 MiB."
            );
        }

        try {
            byte[] bytes = file.getBytes();

            if (bytes.length == 0
                    || bytes.length > MAX_FILE_SIZE) {
                throw new IllegalArgumentException(
                        "El tamaño de la imagen no es válido."
                );
            }

            // Validar que sea una imagen real admitida.
            ImageFormat format = validateImage(bytes);

            // Generar identificador independiente
            // del nombre original enviado por el cliente.
            String publicId =
                    UUID.randomUUID().toString();

            Map<String, Object> options = ObjectUtils.asMap(
                    "folder", "catalogo-digital/products",
                    "public_id", publicId,
                    "resource_type", "image",
                    "overwrite", false
            );

            // Subir a Cloudinary.
            Map<?, ?> result = cloudinary.uploader()
                    .upload(bytes, options);

            Object returnedPublicId = result.get("public_id");
            Object secureUrl = result.get("secure_url");

            if (!(returnedPublicId instanceof String)
                    || !(secureUrl instanceof String)) {
                throw new IllegalStateException(
                        "Cloudinary devolvió una respuesta incompleta."
                );
            }

            String imageUrl = (String) secureUrl;

            if (!imageUrl.startsWith("https://")) {
                throw new IllegalStateException(
                        "Cloudinary no devolvió una URL HTTPS."
                );
            }

            return new StoredImage(
                    (String) returnedPublicId,
                    imageUrl,
                    format.contentType(),
                    bytes.length
            );

        } catch (IOException exception) {
            throw new IllegalStateException(
                    "No se pudo subir la imagen a Cloudinary.",
                    exception
            );
        }
    }

    // 2. Validar contenido y dimensiones
    private ImageFormat validateImage(byte[] bytes)
            throws IOException {

        ImageFormat expected = detectFormat(bytes);

        try (ImageInputStream stream =
                     ImageIO.createImageInputStream(
                             new ByteArrayInputStream(bytes)
                     )) {

            if (stream == null) {
                throw new IllegalArgumentException(
                        "El archivo no es una imagen válida."
                );
            }

            Iterator<ImageReader> readers =
                    ImageIO.getImageReaders(stream);

            if (!readers.hasNext()) {
                throw new IllegalArgumentException(
                        "No se pudo interpretar la imagen."
                );
            }

            ImageReader reader = readers.next();

            try {
                reader.setInput(stream);

                String detectedFormat =
                        reader.getFormatName();

                boolean matches =
                        expected.extension().equals("png")
                                ? "png".equalsIgnoreCase(detectedFormat)
                                : "jpeg".equalsIgnoreCase(detectedFormat)
                                  || "jpg".equalsIgnoreCase(detectedFormat);

                if (!matches) {
                    throw new IllegalArgumentException(
                            "El formato de la imagen no coincide."
                    );
                }

                int width = reader.getWidth(0);
                int height = reader.getHeight(0);

                if (width <= 0 || height <= 0
                        || width > MAX_DIMENSION
                        || height > MAX_DIMENSION) {
                    throw new IllegalArgumentException(
                            "Las dimensiones de la imagen no son válidas."
                    );
                }

                if (reader.read(0) == null) {
                    throw new IllegalArgumentException(
                            "No se pudo decodificar la imagen."
                    );
                }

            } finally {
                reader.dispose();
            }
        }

        return expected;
    }

    // 3. Detectar PNG o JPEG por la firma
    private ImageFormat detectFormat(byte[] bytes) {

        // PNG
        if (bytes.length >= 8
                && (bytes[0] & 0xFF) == 0x89
                && bytes[1] == 0x50
                && bytes[2] == 0x4E
                && bytes[3] == 0x47
                && bytes[4] == 0x0D
                && bytes[5] == 0x0A
                && bytes[6] == 0x1A
                && bytes[7] == 0x0A) {

            return new ImageFormat("png", "image/png");
        }

        // JPEG
        if (bytes.length >= 3
                && (bytes[0] & 0xFF) == 0xFF
                && (bytes[1] & 0xFF) == 0xD8
                && (bytes[2] & 0xFF) == 0xFF) {

            return new ImageFormat("jpg", "image/jpeg");
        }

        throw new IllegalArgumentException(
                "Solo se permiten imágenes PNG y JPEG."
        );
    }

    private record ImageFormat(
            String extension,
            String contentType
    ) {
    }

    // Información devuelta después de la subida
    public record StoredImage(
            String publicId,
            String secureUrl,
            String contentType,
            long size
    ) {
    }


public void delete(String publicId) {

    if (publicId == null || publicId.isBlank()) {
        throw new IllegalArgumentException(
                "El identificador de Cloudinary es obligatorio."
        );
    }

    try {
        Map<?, ?> result = cloudinary.uploader().destroy(
                publicId,
                ObjectUtils.asMap(
                        "resource_type", "image",
                        "invalidate", true
                )
        );

        Object status = result.get("result");

        if (!"ok".equals(status)
                && !"not found".equals(status)) {
            throw new IllegalStateException(
                    "Cloudinary no confirmó la eliminación."
            );
        }

    } catch (IOException exception) {
        throw new IllegalStateException(
                "No se pudo eliminar el archivo de Cloudinary.",
                exception
        );
    }
}
